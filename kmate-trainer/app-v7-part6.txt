class StockfishPlayEngine {
  constructor(workerUrl) {
    this.worker = new Worker(workerUrl);
    this.waiters = new Set();
    this.searchQueue = Promise.resolve();
    this.lastInfo = {};
    this.isReady = false;
    this.worker.addEventListener('message', (event) => this.handlePayload(event.data));
    this.worker.addEventListener('error', (event) => {
      this.rejectAll(new Error(event.message || 'Stockfish worker error'));
    });
    this.ready = this.initialize();
  }

  send(command) {
    this.worker.postMessage(command);
  }

  handlePayload(payload) {
    const lines = String(payload ?? '').split(/\r?\n/).map((line) => line.trim()).filter(Boolean);
    for (const line of lines) {
      if (line.startsWith('info ')) {
        const depth = line.match(/\bdepth\s+(\d+)/)?.[1];
        const nodes = line.match(/\bnodes\s+(\d+)/)?.[1];
        const cp = line.match(/\bscore\s+cp\s+(-?\d+)/)?.[1];
        const mate = line.match(/\bscore\s+mate\s+(-?\d+)/)?.[1];
        if (depth) this.lastInfo.depth = Number(depth);
        if (nodes) this.lastInfo.nodes = Number(nodes);
        if (cp) this.lastInfo.scoreCp = Number(cp);
        if (mate) this.lastInfo.mate = Number(mate);
      }
      for (const waiter of [...this.waiters]) {
        let matched = false;
        try {
          matched = waiter.predicate(line);
        } catch (error) {
          this.waiters.delete(waiter);
          window.clearTimeout(waiter.timer);
          waiter.reject(error);
          continue;
        }
        if (matched) {
          this.waiters.delete(waiter);
          window.clearTimeout(waiter.timer);
          waiter.resolve(line);
        }
      }
    }
  }

  waitFor(predicate, timeoutMs = 20000) {
    return new Promise((resolve, reject) => {
      const waiter = { predicate, resolve, reject, timer: null };
      waiter.timer = window.setTimeout(() => {
        this.waiters.delete(waiter);
        reject(new Error('Stockfish response timed out'));
      }, timeoutMs);
      this.waiters.add(waiter);
    });
  }

  rejectAll(error) {
    for (const waiter of this.waiters) {
      window.clearTimeout(waiter.timer);
      waiter.reject(error);
    }
    this.waiters.clear();
  }

  async initialize() {
    const uciOk = this.waitFor((line) => line === 'uciok', 45000);
    this.send('uci');
    await uciOk;
    this.send('setoption name Hash value 16');
    this.send('setoption name Threads value 1');
    this.send('setoption name Ponder value false');
    this.send('setoption name MultiPV value 1');
    const readyOk = this.waitFor((line) => line === 'readyok', 30000);
    this.send('isready');
    await readyOk;
    this.isReady = true;
    return true;
  }

  async configureRating(rating) {
    const requested = Math.max(1000, Math.min(2200, Number(rating) || 1600));
    if (requested >= STOCKFISH_MIN_ELO) {
      this.send('setoption name Skill Level value 20');
      this.send('setoption name UCI_LimitStrength value true');
      this.send(`setoption name UCI_Elo value ${Math.max(STOCKFISH_MIN_ELO, Math.min(STOCKFISH_MAX_ELO, requested))}`);
    } else {
      const skill = Math.max(0, Math.min(4, Math.round((requested - 1000) / 65)));
      this.send('setoption name UCI_LimitStrength value false');
      this.send(`setoption name Skill Level value ${skill}`);
    }
    this.send('setoption name MultiPV value 1');
    const readyOk = this.waitFor((line) => line === 'readyok', 15000);
    this.send('isready');
    await readyOk;
  }

  bestMove({ fen, rating, movetime }) {
    const run = async () => {
      await this.ready;
      await this.configureRating(rating);
      this.lastInfo = {};
      this.send(`position fen ${fen}`);
      const bestMoveLine = this.waitFor((line) => line.startsWith('bestmove '), Math.max(15000, movetime + 12000));
      this.send(`go movetime ${Math.max(120, Math.round(movetime || 600))}`);
      const line = await bestMoveLine;
      const move = line.split(/\s+/)[1];
      if (!move || move === '(none)' || move === '0000') throw new Error(`Stockfish returned no legal move: ${line}`);
      return { move, ...this.lastInfo };
    };
    const result = this.searchQueue.then(run, run);
    this.searchQueue = result.catch(() => undefined);
    return result;
  }

  stop() {
    try {
      this.send('stop');
    } catch {}
  }
}

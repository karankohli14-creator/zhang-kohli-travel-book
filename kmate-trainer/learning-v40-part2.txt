function ensureLearningNav() {
  const nav = $('.topnav');
  if (!nav) return null;
  let button = $('#learningNavButton');
  if (!button) {
    button = document.createElement('button');
    button.id = 'learningNavButton';
    button.className = 'navbutton';
    button.type = 'button';
    button.dataset.view = 'learning';
    button.textContent = 'Learn';
    button.addEventListener('click', openLearningView);
    nav.append(button);
  }
  for (const existing of $$('.navbutton', nav)) {
    if (existing === button || existing.dataset.learningHideBound) continue;
    existing.dataset.learningHideBound = LEARNING_VERSION;
    existing.addEventListener('click', () => {
      const view = $('#learningView');
      if (view) view.hidden = true;
    }, { capture: true });
  }
  return button;
}

function openLearningView() {
  const state = window.__KMATE__?.state?.();
  if (state && !state.finalized && !$('#gameView')?.hidden) {
    showToast('Finish the current position or return to setup before opening Learning.');
    return false;
  }
  const view = ensureLearningView();
  $('#setupView').hidden = true;
  $('#gameView').hidden = true;
  $('#insightsView').hidden = true;
  view.hidden = false;
  document.documentElement.classList.remove('game-mode');
  document.body.classList.remove('game-mode', 'board-focus', 'game-panel-open');
  $$('.navbutton').forEach((button) => button.classList.toggle('active', button.dataset.view === 'learning'));
  window.scrollTo({ top: 0, left: 0, behavior: 'auto' });
  void renderLearningView();
  return true;
}

function bindLearningViewControls(view) {
  $('#learningStartPuzzles', view)?.addEventListener('click', () => startRecommendedWorkout());
  $('#learningWatchLesson', view)?.addEventListener('click', () => openRecommendedVideo());
  $('#learningRetryPosition', view)?.addEventListener('click', retryExactPosition);
}

function focusBrowser(recommendation) {
  const container = $('#learningFocusBrowser');
  const taxonomy = core()?.taxonomy || {};
  if (!container) return;
  const active = dashboardFocusOverride || recommendation?.focus?.key || 'calculation';
  container.innerHTML = Object.values(taxonomy).map((focus) => `
    <button class="learning-focus-choice ${focus.key === active ? 'active' : ''}" type="button" data-learning-focus="${escapeHtml(focus.key)}">
      <span>${escapeHtml(focus.icon)}</span><b>${escapeHtml(focus.label)}</b>
    </button>`).join('');
  for (const button of $$('[data-learning-focus]', container)) {
    button.addEventListener('click', () => {
      dashboardFocusOverride = button.dataset.learningFocus === recommendation?.focus?.key ? null : button.dataset.learningFocus;
      void renderLearningView();
    });
  }
}

function profileMarkup(profile) {
  if (!profile?.length || profile.every((item) => item.score <= 0)) {
    return '<div class="learning-profile-row"><span>◎</span><div><b>Build your baseline</b><small>Complete a few positions so repeated signals can emerge.</small></div><em class="learning-pattern">baseline</em></div>';
  }
  return profile.slice(0, 4).map((item) => `
    <div class="learning-profile-row">
      <span>${escapeHtml(item.icon)}</span>
      <div><b>${escapeHtml(item.label)}</b><small>${item.occurrences ? `${item.occurrences} recent session${item.occurrences === 1 ? '' : 's'}` : 'Contextual focus'}</small></div>
      <em class="learning-pattern ${escapeHtml(item.pattern)}">${escapeHtml(item.pattern)}</em>
    </div>`).join('');
}

function videoCardMarkup(video, catalog) {
  if (!video) return '<div class="learning-video-copy"><small>Lesson unavailable</small><b>Use the puzzle workout for now</b><span>The external video catalog could not be loaded.</span></div>';
  const label = catalog?.contentPolicy?.labels?.[video.licenseClass] || 'External lesson';
  return `
    <div class="learning-video-thumb" aria-hidden="true"></div>
    <div class="learning-video-copy">
      <small>${escapeHtml(label)}</small>
      <b>${escapeHtml(video.title)}</b>
      <span>${escapeHtml(video.creator)} · ${escapeHtml(video.level)}</span>
      <div class="learning-video-actions">
        <button class="learning-mini-button primary-lite" type="button" data-open-learning-video="${escapeHtml(video.id)}">Watch lesson</button>
        <a class="learning-mini-button" href="${escapeHtml(video.sourceUrl)}" target="_blank" rel="noopener noreferrer">Open on Lichess</a>
      </div>
    </div>`;
}

async function renderLearningView() {
  const recommendation = safeRecommendation();
  if (!recommendation) return;
  const taxonomy = recommendation.focus;
  $('#learningFocusTitle').textContent = `Train ${taxonomy.label.toLowerCase()} next.`;
  $('#learningFocusDescription').textContent = taxonomy.description;
  $('#learningFocusIcon').textContent = taxonomy.icon;
  $('#learningFocusLabel').textContent = taxonomy.label;
  $('#learningFocusBand').textContent = `around ${recommendation.puzzleRating}`;
  $('#learningReason').textContent = (recommendation.reasons || []).slice(0, 2).join(' ');
  $('#learningProfile').innerHTML = profileMarkup(recommendation.profile);
  focusBrowser(recommendation);

  const stats = learningStats();
  $('#learningSolvedCount').textContent = String(stats.solved);
  $('#learningWorkoutCount').textContent = String(stats.workouts);
  $('#learningVideoCount').textContent = String(stats.videos);
  $('#learningStreakCount').textContent = String(stats.streak);
  const retry = $('#learningRetryPosition');
  if (retry) retry.disabled = !recommendation.exactRetryAvailable;

  try {
    const catalog = await loadVideoCatalog();
    const video = chooseVideo(catalog, recommendation);
    const card = $('#learningVideoCard');
    if (card) {
      card.innerHTML = videoCardMarkup(video, catalog);
      $('[data-open-learning-video]', card)?.addEventListener('click', () => openVideo(video));
    }
    $('#learningWatchLesson').disabled = !video;
    $('#learningWatchLesson').dataset.videoId = video?.id || '';
  } catch (error) {
    console.warn('K-Mate video catalog failed to load.', error);
    $('#learningVideoCard').innerHTML = videoCardMarkup(null, null);
    $('#learningWatchLesson').disabled = true;
  }

  try {
    const index = await loadPuzzleIndex();
    const button = $('#learningStartPuzzles');
    if (button) button.textContent = `Start 5 puzzles · ${Number(index.total || 0).toLocaleString()} CC0 available`;
  } catch {
    const button = $('#learningStartPuzzles');
    if (button) button.textContent = 'Start 5 recommended puzzles';
  }
}

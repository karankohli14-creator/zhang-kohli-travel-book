
/* K-Mate v40 — make Learning reachable while the full-screen setup wizard is active. */
function ensureWizardLearningButton() {
  const footer = $('#setupWizard .wizard-welcome-footer');
  if (!footer) return null;
  footer.classList.add('kmate-learning-footer');

  if (!$('#kmateLearningWizardStyles')) {
    const style = document.createElement('style');
    style.id = 'kmateLearningWizardStyles';
    style.textContent = `
      .wizard-welcome-footer.kmate-learning-footer.kmate-import-footer{
        grid-template-columns:.76fr .76fr .76fr 1.3fr;
      }
      .wizard-welcome-footer.kmate-learning-footer:not(.kmate-import-footer){
        grid-template-columns:.85fr .85fr 1.35fr;
      }
      #wizardLearningButton{border-color:#80d8a44d;background:#80d8a40b;color:#c8f7da}
      @media(max-width:760px){
        .wizard-welcome-footer.kmate-learning-footer.kmate-import-footer{
          grid-template-columns:.72fr .72fr .72fr 1.15fr;
        }
        #wizardLearningButton{padding-inline:6px;gap:4px;font-size:11px}
      }
      @media(max-width:430px){
        #wizardLearningButton span{display:none}
        .wizard-welcome-footer.kmate-learning-footer.kmate-import-footer{
          grid-template-columns:.68fr .68fr .68fr 1.08fr;
        }
      }
    `;
    document.head.append(style);
  }

  let button = $('#wizardLearningButton');
  if (button) return button;
  button = document.createElement('button');
  button.id = 'wizardLearningButton';
  button.type = 'button';
  button.className = 'wizard-button wizard-secondary';
  button.setAttribute('aria-label', 'Open adaptive learning recommendations');
  button.innerHTML = '<span aria-hidden="true">◈</span><b>Learn</b>';
  button.addEventListener('click', openLearningView);
  const primary = footer.querySelector('[data-wizard-next="position"]');
  footer.insertBefore(button, primary || null);
  return button;
}

const km40InitializeLearningWithWizard = initializeLearning;
initializeLearning = function initializeLearningV40(attempt = 0) {
  km40InitializeLearningWithWizard(attempt);
  if (core() && $('#learningView')) ensureWizardLearningButton();
};

ensureWizardLearningButton();

ex] = clone(currentSession);
    saveStore();
  }
}

function sessionCoach(session) {
  if (session.reason === 'timeout') return 'Clock management was the decisive issue. Consider 3+2 or 5+0 while building a repeatable thinking rhythm.';
  if ((session.takebacks || 0) >= 2) return 'Frequent takebacks suggest candidate-move uncertainty. Compare two serious candidates before touching a piece.';
  if (Number.isFinite(session.avgCpLoss) && session.avgCpLoss > 180) return `The largest signal is decision quality in ${session.tags[0] || 'this position'}. Slow the candidate-move scan before forcing moves.`;
  if (Number.isFinite(session.avgCpLoss) && session.avgCpLoss > 100) return `This was an inconsistent session. Focus on ${session.tags[0] || 'the main positional theme'} and use more of the clock on irreversible moves.`;
  if (session.outcome === 'loss') return `The result was difficult, but the move-quality estimate was reasonable. Review the ${session.theme.toLowerCase()} plan rather than only the final tactic.`;
  return 'Solid session. Continue with the same opening at a slightly stronger opponent setting or move to the late middlegame version.';
}

function resign() {
  if (!game || finalized) return;
  requestId += 1;
  thinking = false;
  finalizeSession('loss', 'resignation', true);
  showResult('Position resigned', `This session is recorded as a loss against the ${settings.opponentRating} setting.`, '0');
  renderAll();
}

function takeback() {
  if (!game || thinking || finalized) return;
  syncClock();
  requestId += 1;
  const history = game.history();
  if (!history.length) return;
  let undoneUserMoves = 0;
  if (game.turn() === userColor && history.length >= 2) {
    const engineMove = game.undo();
    const userMove = game.undo();
    if (userMove?.color === userColor) undoneUserMoves = 1;
    if (engineMove?.color === userColor) undoneUserMoves += 1;
  } else {
    const move = game.undo();
    if (move?.color === userColor) undoneUserMoves = 1;
  }
  while (undoneUserMoves > 0 && currentSession.userMoves.length) {
    currentSession.userMoves.pop();
    undoneUserMoves -= 1;
  }
  currentSession.takebacks += 1;
  selected = null;
  legalMoves = [];
  lastMove = null;
  thinking = false;
  beginNextTurn();
  setStatus('Moves taken back. The clock was not restored; try a different plan.');
  renderAll();
  if (game.turn() === engineColor) askEngine();
}

function escapeHtml(value) {
  return String(value ?? '')
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#039;');
}

function completedDetailedSessions() {
  return store.sessions.filter((session) => session.completed && ['win', 'draw', 'loss'].includes(session.outcome));
}

function totalRecord() {
  const detailed = completedDetailedSessions();
  const wins = detailed.filter((session) => session.outcome === 'win').length + store.legacy.wins;
  const draws = detailed.filter((session) => session.outcome === 'draw').length + store.legacy.draws;
  const losses = detailed.filter((session) => session.outcome === 'loss').length + store.legacy.losses;
  const games = wins + draws + losses;
  const best = Math.max(store.legacy.best || 0, ...detailed.filter((session) => session.outcome === 'win').map((session) => session.opponentRating), 0);
  return { games, wins, draws, losses, best, score: games ? (wins + draws * 0.5) / games : null };
}

function groupSessions(sessions, keyFunction) {
  const groups = new Map();
  for (const session of sessions) {
    const key = keyFunction(session);
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(session);
  }
  return groups;
}

function aggregateSessions(sessions) {
  const completed = sessions.filter((session) => session.completed && ['win', 'draw', 'loss'].includes(session.outcome));
  const points = completed.reduce((sum, session) => sum + (scoreForOutcome(session.outcome) ?? 0), 0);
  const moves = sessions.flatMap((session) => session.userMoves || []).filter((move) => Number.isFinite(move.cpLoss));
  const white = completed.filter((session) => session.userColor === 'w');
  const black = completed.filter((session) => session.userColor === 'b');
  const colorScore = (list) => list.length
    ? list.reduce((sum, session) => sum + scoreForOutcome(session.outcome), 0) / list.length
    : null;
  return {
    games: completed.length,
    score: completed.length ? points / completed.length : null,
    avgCpLoss: average(moves.map((move) => move.cpLoss)),
    analyzedMoves: moves.length,
    timeouts: completed.filter((session) => session.reason === 'timeout').length,
    whiteGames: white.length,
    blackGames: black.length,
    whiteScore: colorScore(white),
    blackScore: colorScore(black),
  };
}

function trainingPerformance(sessions) {
  const completed = sessions.filter((session) => session.completed && ['win', 'draw', 'loss'].includes(session.outcome));
  if (completed.length < 3) return null;
  const averageOpponent = average(completed.map((session) => session.opponentRating));
  const score = completed.reduce((sum, session) => sum + scoreForOutcome(session.outcome), 0) / completed.length;
  let adjustment;
  if (score <= 0) adjustment = -600;
  else if (score >= 1) adjustment = 600;
  else adjustment = 400 * Math.log10(score / (1 - score));
  return Math.round(Math.max(800, Math.min(2600, averageOpponent + adjustment)));
}

function weakestGroup(groups, minimumGames = 2) {
  const candidates = [...groups.entries()].map(([name, sessions]) => ({ name, sessions, ...aggregateSessions(sessions) }));
  const qualified = candidates.filter((item) => item.games >= minimumGames);
  const pool = qualified.length ? qualified : candidates.filter((item) => item.games > 0);
  if (!pool.length) return null;
  return pool.sort((a, b) => {
    const weaknessA = (1 - (a.score ?? 0.5)) * 180 + (a.avgCpLoss ?? 90);
    const weaknessB = (1 - (b.score ?? 0.5)) * 180 + (b.avgCpLoss ?? 90);
    return weaknessB - weaknessA;
  })[0];
}

function themeGroups() {
  const groups = new Map();
  for (const session of store.sessions) {
    for (const tag of session.tags || []) {
      if (!groups.has(tag)) groups.set(tag, []);
      groups.get(tag).push(session);
    }
  }
  return groups;
}

function renderSetupSignal() {
  const completed = completedDetailedSessions();
  if (!completed.length) {
    $('#setupSignalTitle').textContent = 'Play a few sessions to build your profile';
    $('#setupSignalText').textContent = 'K-Mate will compare results across openings, phases, themes, colors, opponent ratings, and time controls.';
    return;
  }
  const opening = weakestGroup(groupSessions(completed.filter((session) => session.opening !== 'Various'), (session) => session.opening));
  const phase = weakestGroup(groupSessions(completed, (session) => session.phase));
  if (opening && opening.games >= 2) {
    $('#setupSignalTitle').textContent = `${opening.name} is your clearest current opening signal`;
    $('#setupSignalText').textContent = `${opening.games} games · ${percent(opening.score)} score${Number.isFinite(opening.avgCpLoss) ? ` · ${Math.round(opening.avgCpLoss)} cp average estimated loss` : ''}.`;
  } else if (phase) {
    $('#setupSignalTitle').textContent = `${phaseLabel(phase.name)} is the phase to watch`;
    $('#setupSignalText').textContent = `${phase.games} completed session${phase.games === 1 ? '' : 's'} · ${percent(phase.score)} score. More games will make the signal more reliable.`;
  }
}

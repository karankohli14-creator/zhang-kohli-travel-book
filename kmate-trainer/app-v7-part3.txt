        row.black = move.san;
      }
      number += 1;
    }
  }
  const list = $('#moveList');
  list.innerHTML = rows.length
    ? rows.map((item) => `<div class="moverow"><span>${item.number}.</span><b>${escapeHtml(item.white)}</b><b>${escapeHtml(item.black)}</b></div>`).join('')
    : '<div class="emptyMoves">The move list will appear here.</div>';
  list.scrollTop = list.scrollHeight;
  $('#moveCount').textContent = `${history.length} ${history.length === 1 ? 'ply' : 'plies'}`;
  $('#sessionNote').textContent = history.length >= 40
    ? `You have played ${Math.floor(history.length / 2)} moves from this position. Continue, resign, or draw a new position.`
    : 'The position continues until mate, draw, timeout, resignation, or a new position.';
}

function renderMaterialAdvantage() {
  if (!game) return;
  const values = { p: 1, n: 3, b: 3, r: 5, q: 9, k: 0 };
  const totals = { w: 0, b: 0 };
  game.board().forEach((row) => row.forEach((piece) => {
    if (piece) totals[piece.color] += values[piece.type] || 0;
  }));
  const diff = totals[userColor] - totals[engineColor];
  const userEl = $('#userMaterial');
  const engineEl = $('#engineMaterial');
  if (!userEl || !engineEl) return;
  if (diff > 0) {
    userEl.textContent = `+${diff}`;
    engineEl.textContent = '−';
    userEl.classList.add('ahead'); engineEl.classList.remove('ahead');
  } else if (diff < 0) {
    userEl.textContent = '−';
    engineEl.textContent = `+${Math.abs(diff)}`;
    engineEl.classList.add('ahead'); userEl.classList.remove('ahead');
  } else {
    userEl.textContent = 'Even'; engineEl.textContent = 'Even';
    userEl.classList.remove('ahead'); engineEl.classList.remove('ahead');
  }
  userEl.setAttribute('aria-label', diff > 0 ? `You are up ${diff} material points` : diff < 0 ? `You are down ${Math.abs(diff)} material points` : 'Material is even');
  engineEl.setAttribute('aria-label', diff < 0 ? `Opponent is up ${Math.abs(diff)} material points` : diff > 0 ? `Opponent is down ${diff} material points` : 'Material is even');
}

function tapSquare(square) {
  if (!game || thinking || finalized || game.isGameOver() || game.turn() !== userColor) return;
  const piece = game.get(square);
  if (!selected) {
    if (piece?.color === userColor) {
      selected = square;
      legalMoves = game.moves({ square, verbose: true });
      renderBoard();
    }
    return;
  }
  if (piece?.color === userColor) {
    selected = square;
    legalMoves = game.moves({ square, verbose: true });
    renderBoard();
    return;
  }
  const candidates = legalMoves.filter((move) => move.to === square);
  if (!candidates.length) {
    selected = null;
    legalMoves = [];
    renderBoard();
    return;
  }
  if (candidates.some((move) => move.promotion)) {
    promotionBase = { from: selected, to: square };
    openPromotion();
    return;
  }
  makeUserMove({ from: selected, to: square, promotion: 'q' });
}

function openPromotion() {
  const choices = $('#promotionChoices');
  choices.innerHTML = '';
  for (const type of ['q', 'r', 'b', 'n']) {
    const button = document.createElement('button');
    button.type = 'button';
    const glyph = document.createElement('span');
    glyph.className = `piece ${userColor === 'w' ? 'white' : 'black'}`;
    glyph.textContent = PIECES[type];
    button.append(glyph);
    button.addEventListener('click', () => {
      closeDialog('promotionDialog');
      makeUserMove({ ...promotionBase, promotion: type });
    });
    choices.append(button);
  }
  openDialog('promotionDialog');
}

function uciFromMove(move) {
  return `${move.from}${move.to}${move.promotion || ''}`;
}

function makeUserMove(moveObject) {
  if (!game || finalized || game.turn() !== userColor) return;
  syncClock();
  if (finalized) return;
  const fenBefore = game.fen();
  const spentMs = spentThisTurn(userColor);
  let move;
  try {
    move = game.move(moveObject);
  } catch {
    return;
  }
  applyIncrement(userColor);
  const moveRecord = {
    id: `${currentSession.id}-m${currentSession.userMoves.length + 1}`,
    san: move.san,
    uci: uciFromMove(move),
    spentMs,
    remainingMs: clockRunning ? Math.max(0, clocks[userColor]) : null,
    cpLoss: null,
    bestMove: null,
    quality: 'pending',
  };
  currentSession.userMoves.push(moveRecord);
  requestMoveAnalysis(fenBefore, moveRecord);
  selected = null;
  legalMoves = [];
  lastMove = { from: move.from, to: move.to };
  beginNextTurn();
  setStatus(`${game.isCheck() ? 'Check. ' : ''}Opponent is considering the position.`, 'thinking');
  renderAll();
  if (finishIfNeeded()) return;
  askEngine();
}

function fallbackMove() {
  const moves = game.moves({ verbose: true });
  if (!moves.length) return null;
  moves.sort((a, b) => {
    const score = (move) => (move.captured ? 100 : 0) + (move.promotion ? 500 : 0) + (move.san.includes('+') ? 40 : 0);
    return score(b) - score(a);
  });
  const top = Math.max(1, Math.min(moves.length, settings.opponentRating >= 1900 ? 2 : settings.opponentRating >= 1600 ? 4 : 7));
  return moves[Math.floor(Math.random() * top)];
}

async function askEngine() {
  if (!game || finalized || game.turn() !== engineColor) return;
  thinking = true;
  renderTurns();
  const id = ++requestId;
  const fen = game.fen();
  const baseThinkMs = settings.opponentRating >= 2100
    ? 1400
    : settings.opponentRating >= 1900
      ? 1050
      : settings.opponentRating >= 1700
        ? 760
        : settings.opponentRating >= 1500
          ? 560
          : 360;
  const maxMs = clockRunning
    ? Math.max(180, Math.min(baseThinkMs, clocks[engineColor] * 0.035))
    : baseThinkMs;

  if (!stockfishEngine) {
    window.setTimeout(() => applyEngineReply(id, { move: null, fallback: true }), 250);
    return;
  }

  try {
    const result = await stockfishEngine.bestMove({
      fen,
      rating: settings.opponentRating,
      movetime: Math.round(maxMs),
    });
    applyEngineReply(id, { task: 'move', id, ...result, source: 'stockfish-18' });
  } catch (error) {
    console.warn('Stockfish move failed; using emergency fallback.', error);
    applyEngineReply(id, { task: 'move', id, move: null, fallback: true, error: String(error) });
  }
}

function applyEngineReply(id, data) {
  if (id !== requestId || !game || finalized || game.isGameOver() || game.turn() !== engineColor) return;
  syncClock();
  if (finalized) return;
  let move = null;
  if (data.move) {
    try {
      move = game.move({
        from: data.move.slice(0, 2),
        to: data.move.slice(2, 4),
        promotion: data.move[4] || 'q',
      });
    } catch {}
  }
  if (!move) {
    const fallback = fallbackMove();
    if (fallback) {
      try {
        move = game.move({ from: fallback.from, to: fallback.to, promotion: fallback.promotion || 'q' });
      } catch {}
    }
  }
  thinking = false;
  if (!move) {
    finishIfNeeded();
    return;
  }
  applyIncrement(engineColor);
  lastMove = { from: move.from, to: move.to };
  beginNextTurn();
  setStatus(game.isCheck() ? 'Check—your king is under attack.' : 'Your move. Build the position one decision at a time.');
  renderAll();
  finishIfNeeded();
}

function requestMoveAnalysis(fenBefore, moveRecord) {
  if (!analysisWorker) return;
  const id = ++analysisRequestId;
  const sessionId = currentSession.id;
  const moveId = moveRecord.id;
  analysisWorker.postMessage({
    task: 'analyze',
    id,
    fen: fenBefore,
    move: moveRecord.uci,
    phase: current.phase,
    perspective: userColor,
  });
  const handler = (event) => {
    if (event.data.task !== 'analyze' || event.data.id !== id) return;
    analysisWorker.removeEventListener('message', handler);
    const targetSession = currentSession?.id === sessionId
      ? currentSession
      : store.sessions.find((session) => session.id === sessionId);
    const targetMove = targetSession?.userMoves?.find((move) => move.id === moveId);
    if (!targetMove || !Number.isFinite(event.data.cpLoss)) return;
    targetMove.cpLoss = event.data.cpLoss;
    targetMove.bestMove = event.data.bestMove || null;
    targetMove.quality = qualityForLoss(event.data.cpLoss).key;
    if (currentSession?.id === sessionId) {
      renderLiveQuality();
      updateStoredCurrentSession();
    } else {
      saveStore();
    }
  };
  analysisWorker.addEventListener('message', handler);
}

function finishIfNeeded() {
  if (!game?.isGameOver() || finalized) return false;
  let outcome;
  let title;
  let text;
  let symbol;
  if (game.isCheckmate()) {
    const winner = game.turn() === 'w' ? 'b' : 'w';
    outcome = winner === userColor ? 'win' : 'loss';
    title = outcome === 'win' ? 'You converted the position' : 'The opponent converted';
    text = outcome === 'win'
      ? `Checkmate against the ${settings.opponentRating} strength setting.`
      : 'You were checkmated. Review the phase, opening, and theme signals before replaying.';
    symbol = outcome === 'win' ? '1' : '0';
  } else {
    outcome = 'draw';
    title = 'Draw';
    text = 'The position ended in a draw. Holding a difficult position is valuable practice.';
    symbol = '½';
  }
  finalizeSession(outcome, game.isCheckmate() ? 'checkmate' : 'draw', true);
  showResult(title, text, symbol);
  renderAll();
  return true;
}

function summarizeSession(session) {
  const losses = session.userMoves.map((move) => move.cpLoss).filter(Number.isFinite);
  const quality = { excellent: 0, good: 0, inaccuracy: 0, mistake: 0, blunder: 0 };
  for (const move of session.userMoves) {
    if (quality[move.quality] !== undefined) quality[move.quality] += 1;
  }
  const control = TIME_CONTROLS[session.timeControl] || TIME_CONTROLS.untimed;
  const used = control.base
    ? Math.max(0, Math.min(1.5, (control.base * 1000 - (session.remainingMs ?? 0)) / (control.base * 1000)))
    : null;
  return {
    analyzedMoves: losses.length,
    avgCpLoss: average(losses),
    quality,
    avgMoveTimeMs: average(session.userMoves.map((move) => move.spentMs)),
    timeUsedPct: used,
  };
}

function finalizeSession(outcome, reason, completed) {
  if (!currentSession || finalized) return;
  syncClock();
  finalized = true;
  thinking = false;
  requestId += 1;
  stopClock();
  currentSession.outcome = outcome;
  currentSession.reason = reason;
  currentSession.completed = completed;
  currentSession.endedAt = new Date().toISOString();
  currentSession.remainingMs = clockRunning ? Math.max(0, clocks[userColor]) : null;
  Object.assign(currentSession, summarizeSession(currentSession));
  if (completed || currentSession.userMoves.length > 0) {
    store.sessions.unshift(clone(currentSession));
    saveStore();
  }
  renderTurns();
  renderClocks();
}

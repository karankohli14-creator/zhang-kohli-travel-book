function ensureAudioContext() {
  if (!soundEnabled()) return null;
  const AudioCtx = window.AudioContext || window.webkitAudioContext;
  if (!AudioCtx) return null;
  if (!audioContext) audioContext = new AudioCtx();
  if (audioContext.state === 'suspended') audioContext.resume().catch(() => {});
  return audioContext;
}

function playMoveSound(kind = 'move') {
  if (!soundEnabled()) return;
  const ctx = ensureAudioContext();
  if (!ctx) return;
  const now = ctx.currentTime;
  const gain = ctx.createGain();
  const osc = ctx.createOscillator();
  const isCapture = kind === 'capture';
  const isCheck = kind === 'check';
  osc.type = 'sine';
  osc.frequency.setValueAtTime(isCapture ? 190 : isCheck ? 520 : 310, now);
  osc.frequency.exponentialRampToValueAtTime(isCapture ? 135 : isCheck ? 650 : 245, now + (isCapture ? 0.075 : 0.055));
  gain.gain.setValueAtTime(0.0001, now);
  gain.gain.exponentialRampToValueAtTime(isCapture ? 0.085 : 0.055, now + 0.008);
  gain.gain.exponentialRampToValueAtTime(0.0001, now + (isCapture ? 0.105 : 0.08));
  osc.connect(gain);
  gain.connect(ctx.destination);
  osc.start(now);
  osc.stop(now + (isCapture ? 0.11 : 0.085));
}

function formatClock(milliseconds) {
  if (!clockRunning) return '∞';
  const safe = Math.max(0, milliseconds);
  if (safe < 10000) {
    const seconds = Math.floor(safe / 1000);
    const tenths = Math.floor((safe % 1000) / 100);
    return `0:${String(seconds).padStart(2, '0')}.${tenths}`;
  }
  const totalSeconds = Math.ceil(safe / 1000);
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = totalSeconds % 60;
  return `${minutes}:${String(seconds).padStart(2, '0')}`;
}

function shortFen(fen) { return String(fen).split(/\s+/).slice(0, 4).join(' '); }

function materialInfo(g) {
  const x = { pieces: 0, queens: 0 };
  g.board().forEach((row) => row.forEach((p) => { if (p) { x.pieces += 1; if (p.type === 'q') x.queens += 1; } }));
  return x;
}
function phaseFits(g, phase) {
  if (g.isGameOver() || g.moves().length < (phase === 'endgame' ? 2 : 7)) return false;
  const x = materialInfo(g);
  if (phase === 'middlegame') return x.pieces >= 21 && x.queens >= 1;
  if (phase === 'late-middlegame') return x.pieces >= 14;
  return x.pieces <= 20;
}
function quickEval(g, color, phase) {
  if (g.isCheckmate()) return g.turn() === color ? -99999 : 99999;
  if (g.isDraw()) return 0;
  const v = { p: 100, n: 320, b: 335, r: 510, q: 930, k: 0 };
  let score = 0;
  g.board().forEach((row, rank) => row.forEach((p, file) => {
    if (!p) return;
    const sign = p.color === color ? 1 : -1;
    const center = 7 - (Math.abs(3.5 - file) + Math.abs(3.5 - rank));
    let value = v[p.type];
    if (p.type === 'p') value += (p.color === 'w' ? 6 - rank : rank - 1) * (phase === 'endgame' ? 13 : 7);
    if (p.type === 'n' || p.type === 'b') value += Math.max(0, center * 4);
    if (p.type === 'k' && phase === 'endgame') value += Math.max(0, center * 7);
    score += sign * value;
  }));
  if (g.isCheck()) score += g.turn() === color ? -24 : 24;
  return score;
}
function movePriority(m) {
  const v = { p: 100, n: 320, b: 335, r: 510, q: 930, k: 0 };
  return (m.captured ? v[m.captured] + 50 : 0) + (m.promotion ? v[m.promotion] + 250 : 0) + (m.san?.includes('#') ? 99999 : m.san?.includes('+') ? 80 : 0);
}
function generatedMove(g, rating, phase) {
  const color = g.turn();
  const moves = g.moves({ verbose: true });
  const scored = moves.map((m) => {
    g.move({ from: m.from, to: m.to, promotion: m.promotion || 'q' });
    let score = quickEval(g, color, phase) + movePriority(m) * 0.12;
    const replies = g.moves({ verbose: true }).sort((a, b) => movePriority(b) - movePriority(a)).slice(0, rating >= 1800 ? 5 : 3);
    if (replies.length && !g.isGameOver()) {
      let worst = Infinity;
      for (const reply of replies) {
        g.move({ from: reply.from, to: reply.to, promotion: reply.promotion || 'q' });
        worst = Math.min(worst, quickEval(g, color, phase));
        g.undo();
      }
      score = score * 0.45 + worst * 0.55;
    }
    g.undo();
    return { m, score };
  }).sort((a, b) => b.score - a.score);
  const choices = rating >= 2000 ? 2 : rating >= 1800 ? 3 : rating >= 1600 ? 4 : rating >= 1400 ? 6 : 8;
  const pool = scored.slice(0, Math.min(choices, scored.length));
  const bias = rating >= 1900 ? 2.8 : rating >= 1600 ? 1.8 : 1.1;
  const index = Math.min(pool.length - 1, Math.floor(Math.pow(Math.random(), bias) * pool.length));
  return pool[index]?.m || moves[0];
}
function materialPointBalance(g) {
  const values = { p: 1, n: 3, b: 3, r: 5, q: 9, k: 0 };
  const totals = { w: 0, b: 0 };
  g.board().forEach((row) => row.forEach((piece) => {
    if (piece) totals[piece.color] += values[piece.type] || 0;
  }));
  return totals.w - totals.b;
}

function positionPassesQualityGate(g, phase, goalKey) {
  if (!phaseFits(g, phase)) return false;
  const legalCount = g.moves().length;
  if (legalCount < (phase === 'endgame' ? 2 : 6)) return false;
  const imbalance = Math.abs(materialPointBalance(g));
  const tolerance = ['convert', 'defend'].includes(goalKey) ? 6 : 4;
  if (imbalance > tolerance) return false;
  const evaluation = Math.abs(quickEval(g, 'w', phase));
  return evaluation < (['convert', 'defend'].includes(goalKey) ? 780 : 520);
}

function freshPosition() {
  let anchors = candidatePositions();
  if (!anchors.length) anchors = validPositions.filter((p) => p.phase === settings.phase);
  if (!anchors.length) anchors = validPositions;
  const nearest = Math.min(...anchors.map((p) => Math.abs(p.rating - settings.positionRating)));
  let pool = anchors.filter((p) => Math.abs(p.rating - settings.positionRating) <= Math.max(100, nearest));
  if (current?.seedId && pool.length > 1) pool = pool.filter((p) => p.id !== current.seedId);
  for (let attempt = 0; attempt < 28; attempt += 1) {
    const anchor = pool[Math.floor(Math.random() * pool.length)];
    const g = new Chess(anchor.fen);
    const ranges = settings.phase === 'endgame' ? [1, 7] : settings.phase === 'late-middlegame' ? [3, 11] : [3, 9];
    const plies = ranges[0] + Math.floor(Math.random() * (ranges[1] - ranges[0] + 1));
    const line = [];
    for (let ply = 0; ply < plies && !g.isGameOver(); ply += 1) {
      const m = generatedMove(g, Math.max(1100, Math.min(2200, settings.positionRating + Math.round((Math.random() - 0.5) * 240))), settings.phase);
      if (!m) break;
      line.push(g.move({ from: m.from, to: m.to, promotion: m.promotion || 'q' }).san);
    }
    const fen = g.fen();
    if (!genSeen.includes(shortFen(fen)) && positionPassesQualityGate(g, settings.phase, settings.trainingGoal)) {
      rememberFen(fen);
      const code = Math.random().toString(36).slice(2, 6).toUpperCase();
      return { ...anchor, id: `generated-${anchor.id}-${Date.now()}-${code}`, seedId: anchor.id, generated: true, variationPlies: line.length, variation: line, fen, rating: settings.positionRating, title: `${anchor.opening === 'Various' ? anchor.title : anchor.opening} · ${code}`, description: `Quality-gated continuation from “${anchor.title}.” K-Mate varied ${line.length} plies while preserving the opening family, phase, and selected training goal.` };
    }
  }
  const anchor = pool[Math.floor(Math.random() * pool.length)];
  return { ...anchor, id: `generated-${anchor.id}-${Date.now()}`, seedId: anchor.id, generated: false, variationPlies: 0, rating: settings.positionRating };
}

function candidatePositions() {
  let pool = validPositions.filter((position) => position.phase === settings.phase);
  if (settings.phase !== 'endgame' && settings.opening !== 'all') {
    const openingPool = pool.filter((position) => position.opening === settings.opening);
    if (openingPool.length) pool = openingPool;
  }
  const goal = TRAINING_GOALS[settings.trainingGoal] || TRAINING_GOALS.all;
  if (settings.trainingGoal !== 'all') {
    const goalPool = pool.filter((position) => (position.tags || []).some((tag) => goal.tags?.includes(tag)));
    if (goalPool.length) pool = goalPool;
  }
  return pool;
}

}

function chooseCalibrationRating(target) {
  const nearest = CALIBRATION_RATINGS.reduce((best, value) => Math.abs(value - target) < Math.abs(best - target) ? value : best, CALIBRATION_RATINGS[0]);
  const index = CALIBRATION_RATINGS.indexOf(nearest);
  const candidates = CALIBRATION_RATINGS.slice(Math.max(0, index - 2), Math.min(CALIBRATION_RATINGS.length, index + 3));
  return candidates[Math.floor(Math.random() * candidates.length)] || nearest;
}

function initClocks() {
  stopClock();
  const control = timeControl();
  clockRunning = control.base > 0;
  clocks = {
    w: control.base * 1000,
    b: control.base * 1000,
  };
  clockLast = performance.now();
  turnStartRemaining = clocks[game.turn()];
  if (clockRunning) {
    clockInterval = window.setInterval(tickClock, 100);
  }
  renderClocks();
}

function stopClock() {
  if (clockInterval) window.clearInterval(clockInterval);
  clockInterval = null;
}

function syncClock() {
  if (!clockRunning || !game || finalized || game.isGameOver()) return;
  const now = performance.now();
  const active = game.turn();
  const elapsed = Math.max(0, now - clockLast);
  clocks[active] -= elapsed;
  clockLast = now;
  if (clocks[active] <= 0) {
    clocks[active] = 0;
    handleTimeout(active);
  }
}

function tickClock() {
  syncClock();
  renderClocks();
}

function applyIncrement(color) {
  if (!clockRunning) return;
  clocks[color] += timeControl().increment * 1000;
}

function beginNextTurn() {
  clockLast = performance.now();
  turnStartRemaining = clocks[game.turn()];
  renderClocks();
}

function spentThisTurn(color) {
  if (!clockRunning || game.turn() !== color) return null;
  return Math.max(0, turnStartRemaining - clocks[color]);
}

function renderClocks() {
  const userMs = clocks[userColor] ?? 0;
  const engineMs = clocks[engineColor] ?? 0;
  renderClockElement($('#userClock'), userMs, game && !finalized && game.turn() === userColor);
  renderClockElement($('#engineClock'), engineMs, game && !finalized && game.turn() === engineColor);
}

function renderClockElement(element, milliseconds, active) {
  element.textContent = formatClock(milliseconds);
  element.classList.toggle('active', Boolean(active));
  element.classList.toggle('low', clockRunning && milliseconds <= 30000 && milliseconds > 10000);
  element.classList.toggle('critical', clockRunning && milliseconds <= 10000);
}

function hasMatingMaterial(color) {
  if (!game) return true;
  const pieces = [];
  for (let rank = 1; rank <= 8; rank += 1) {
    for (const file of FILES) {
      const piece = game.get(`${file}${rank}`);
      if (piece?.color === color && piece.type !== 'k') pieces.push(piece.type);
    }
  }
  if (pieces.some((piece) => ['p', 'r', 'q'].includes(piece))) return true;
  const bishops = pieces.filter((piece) => piece === 'b').length;
  const knights = pieces.filter((piece) => piece === 'n').length;
  return bishops >= 2 || (bishops >= 1 && knights >= 1) || knights >= 2;
}

function handleTimeout(loser) {
  if (finalized) return;
  requestId += 1;
  thinking = false;
  stopClock();
  const winner = loser === 'w' ? 'b' : 'w';
  const draw = !hasMatingMaterial(winner);
  if (draw) {
    finalizeSession('draw', 'timeout with insufficient mating material', true);
    showResult('Draw on time', 'The flag fell, but the opponent did not have sufficient mating material.', '½');
  } else {
    const outcome = loser === userColor ? 'loss' : 'win';
    finalizeSession(outcome, 'timeout', true);
    showResult(
      outcome === 'win' ? 'You won on time' : 'Time expired',
      outcome === 'win'
        ? `K-Mate ${settings.opponentRating} ran out of time.`
        : `Your ${timeControl().label} clock reached zero.`,
      outcome === 'win' ? '1' : '0',
    );
  }
  renderAll();
}

function orderedSquares() {
  const files = orientation === 'w' ? FILES : [...FILES].reverse().join('');
  const ranks = orientation === 'w' ? '87654321' : '12345678';
  return [...ranks].flatMap((rank) => [...files].map((file) => `${file}${rank}`));
}

function kingInCheckSquare() {
  if (!game?.isCheck()) return null;
  for (let rank = 1; rank <= 8; rank += 1) {
    for (const file of FILES) {
      const square = `${file}${rank}`;
      const piece = game.get(square);
      if (piece?.type === 'k' && piece.color === game.turn()) return square;
    }
  }
  return null;
}

function renderBoard() {
  const board = $('#board');
  board.innerHTML = '';
  if (!game) return;
  const checkSquare = kingInCheckSquare();
  const squares = orderedSquares();
  squares.forEach((square, index) => {
    const piece = game.get(square);
    const button = document.createElement('button');
    button.type = 'button';
    button.className = `sq ${((FILES.indexOf(square[0]) + Number(square[1])) % 2) ? 'light' : 'dark'}`;
    if (selected === square) button.classList.add('selected');
    if (lastMove && (lastMove.from === square || lastMove.to === square)) button.classList.add('last');
    const legal = legalMoves.find((move) => move.to === square);
    if (legal) button.classList.add(piece ? 'capture' : 'legal');
    if (checkSquare === square) button.classList.add('check');
    button.dataset.square = square;
    button.setAttribute('aria-label', `${square}${piece ? ` ${piece.color === 'w' ? 'white' : 'black'} ${piece.type}` : ' empty'}`);

    if (piece) {
      const glyph = document.createElement('span');
      glyph.className = `piece ${piece.color === 'w' ? 'white' : 'black'}`;
      glyph.textContent = PIECES[piece.type];
      button.append(glyph);
    }
    if (index % 8 === 0) {
      const rank = document.createElement('span');
      rank.className = 'coord rank';
      rank.textContent = square[1];
      button.append(rank);
    }
    if (index >= 56) {
      const file = document.createElement('span');
      file.className = 'coord file';
      file.textContent = square[0];
      button.append(file);
    }
    button.addEventListener('click', () => tapSquare(square));
    board.append(button);
  });
}

function moveRatingForRecord(record) {

function renderMoveCell(cell) {
  if (!cell) return '<span class="move-entry move-empty">—</span>';
  if (!cell.isUser) return `<span class="move-entry"><b>${escapeHtml(cell.san)}</b></span>`;
  const rating = moveRatingForRecord(cell.record);
  const cpText = Number.isFinite(cell.record?.cpLoss)
    ? `${Math.round(cell.record.cpLoss)} cp estimated loss`
    : 'Move analysis in progress';
  const bestText = cell.record?.bestMove ? ` · preferred ${cell.record.bestMove}` : '';
  const title = `${rating.label} · ${cpText}${bestText}`;
  return `<span class="move-entry user-move quality-${rating.key}" title="${escapeHtml(title)}"><b>${escapeHtml(cell.san)}</b><span class="move-rating quality-${rating.key}">${escapeHtml(rating.label)}</span></span>`;
}

function renderMoveList() {
  if (!game) return;
  const history = game.history({ verbose: true });
  const userRecords = currentSession?.userMoves || [];
  const rows = [];
  let userRecordIndex = 0;
  let number = startFullmove;
  let row = null;

  for (const move of history) {
    const isUser = move.color === userColor;
    const record = isUser ? (userRecords[userRecordIndex++] || null) : null;
    const cell = { san: move.san, isUser, record };
    if (move.color === 'w') {
      row = { number, white: cell, black: null };
      rows.push(row);
    } else {
      if (!row || row.black) {
        row = { number, white: { san: '…', isUser: false, record: null }, black: cell };
        rows.push(row);
      } else {
        row.black = cell;
      }
      number += 1;
    }
  }

  const list = $('#moveList');
  list.innerHTML = rows.length
    ? rows.map((item) => `<div class="moverow"><span class="move-number">${item.number}.</span>${renderMoveCell(item.white)}${renderMoveCell(item.black)}</div>`).join('')
    : '<div class="emptyMoves">The move list will appear here.</div>';
  list.scrollTop = list.scrollHeight;
  $('#moveCount').textContent = `${history.length} ${history.length === 1 ? 'ply' : 'plies'}`;
  $('#sessionNote').textContent = history.length >= 40
    ? `You have played ${Math.floor(history.length / 2)} moves from this position. Continue, resign, or draw a new position.`
    : 'Your move ratings appear beside your moves as soon as analysis finishes.';
}

function renderMaterialAdvantage() {
  if (!game) return;
  const values = { p: 1, n: 3, b: 3, r: 5, q: 9, k: 0 };
  const totals = { w: 0, b: 0 };
  game.board().forEach((row) => row.forEach((piece) => {
    if (piece) totals[piece.color] += values[piece.type] || 0;
  }));
  const diff = totals[userColor] - totals[engineColor];
  const userEl = $('#userMaterial');
  const engineEl = $('#engineMaterial');
  if (!userEl || !engineEl) return;
  if (diff > 0) {
    userEl.textContent = `+${diff}`;
    engineEl.textContent = '−';
    userEl.classList.add('ahead'); engineEl.classList.remove('ahead');
  } else if (diff < 0) {
    userEl.textContent = '−';
    engineEl.textContent = `+${Math.abs(diff)}`;
    engineEl.classList.add('ahead'); userEl.classList.remove('ahead');
  } else {
    userEl.textContent = 'Even'; engineEl.textContent = 'Even';
    userEl.classList.remove('ahead'); engineEl.classList.remove('ahead');
  }
  userEl.setAttribute('aria-label', diff > 0 ? `You are up ${diff} material points` : diff < 0 ? `You are down ${Math.abs(diff)} material points` : 'Material is even');
  engineEl.setAttribute('aria-label', diff < 0 ? `Opponent is up ${Math.abs(diff)} material points` : diff > 0 ? `Opponent is down ${diff} material points` : 'Material is even');
}

function tapSquare(square) {
  if (!game || thinking || finalized || game.isGameOver() || game.turn() !== userColor) return;
  const piece = game.get(square);
  if (!selected) {
    if (piece?.color === userColor) {
      selected = square;
      legalMoves = game.moves({ square, verbose: true });
      renderBoard();
    }
    return;
  }
  if (piece?.color === userColor) {
    selected = square;
    legalMoves = game.moves({ square, verbose: true });
    renderBoard();
    return;
  }
  const candidates = legalMoves.filter((move) => move.to === square);
  if (!candidates.length) {
    selected = null;
    legalMoves = [];
    renderBoard();
    return;
  }
  if (candidates.some((move) => move.promotion)) {
    promotionBase = { from: selected, to: square };
    openPromotion();
    return;
  }
  makeUserMove({ from: selected, to: square, promotion: 'q' });
}

function openPromotion() {
  const choices = $('#promotionChoices');
  choices.innerHTML = '';
  for (const type of ['q', 'r', 'b', 'n']) {
    const button = document.createElement('button');
    button.type = 'button';
    const glyph = document.createElement('span');
    glyph.className = `piece ${userColor === 'w' ? 'white' : 'black'}`;
    glyph.textContent = PIECES[type];
    button.append(glyph);
    button.addEventListener('click', () => {
      closeDialog('promotionDialog');
      makeUserMove({ ...promotionBase, promotion: type });
    });
    choices.append(button);
  }
  openDialog('promotionDialog');
}

function makeUserMove(moveObject) {
  if (!game || finalized || game.turn() !== userColor) return;
  syncClock();
  if (finalized) return;
  const fenBefore = game.fen();
  const spentMs = spentThisTurn(userColor);
  let move;
  try {
    move = game.move(moveObject);
  } catch {
    return;
  }
  applyIncrement(userColor);
  const moveRecord = {
    id: `${currentSession.id}-m${currentSession.userMoves.length + 1}`,
    san: move.san,
    uci: uciFromMove(move),
    spentMs,
    remainingMs: clockRunning ? Math.max(0, clocks[userColor]) : null,
    cpLoss: null,
    bestMove: null,
    quality: 'pending',
    analysisSource: 'pending',
  };
  currentSession.userMoves.push(moveRecord);
  showMoveQualityBadge(null);
  requestMoveAnalysis(fenBefore, moveRecord);
  selected = null;
  legalMoves = [];
  lastMove = { from: move.from, to: move.to };
  playMoveSound(game.isCheck() ? 'check' : move.captured ? 'capture' : 'move');
  beginNextTurn();
  setStatus(`${game.isCheck() ? 'Check. ' : ''}Opponent is considering the position.`, 'thinking');
  renderAll();
  if (finishIfNeeded()) return;
  askEngine();
}

function fallbackMove() {
  const moves = game.moves({ verbose: true });
  if (!moves.length) return null;
  moves.sort((a, b) => {
    const score = (move) => (move.captured ? 100 : 0) + (move.promotion ? 500 : 0) + (move.san.includes('+') ? 40 : 0);
    return score(b) - score(a);
  });
  const top = Math.max(1, Math.min(moves.length, opponentRatingForSession() >= 1900 ? 2 : opponentRatingForSession() >= 1600 ? 4 : 7));
  return moves[Math.floor(Math.random() * top)];
}

async function askEngine() {
  if (!game || finalized || game.turn() !== engineColor) return;
  thinking = true;
  renderTurns();
  const id = ++requestId;
  const fen = game.fen();
  const sessionRating = opponentRatingForSession();
  const baseThinkMs = sessionRating >= 2100
    ? 1400
    : sessionRating >= 1900
      ? 1050
      : sessionRating >= 1700
        ? 760
        : sessionRating >= 1500
          ? 560
          : 360;
  const maxMs = clockRunning
    ? Math.max(180, Math.min(baseThinkMs, clocks[engineColor] * 0.035))
    : baseThinkMs;

  if (!stockfishEngine) {
    window.setTimeout(() => applyEngineReply(id, { move: null, fallback: true }), 250);
    return;
  }

  try {
    const result = await stockfishEngine.bestMove({
      fen,
      rating: sessionRating,
      movetime: Math.round(maxMs),
    });
    applyEngineReply(id, { task: 'move', id, ...result, source: 'stockfish-18' });
  } catch (error) {
    console.warn('Stockfish move failed; using emergency fallback.', error);
    applyEngineReply(id, { task: 'move', id, move: null, fallback: true, error: String(error) });
  }
}

function applyEngineReply(id, data) {
  if (id !== requestId || !game || finalized || game.isGameOver() || game.turn() !== engineColor) return;
  syncClock();
  if (finalized) return;
  let move = null;
  if (data.move) {
    try {
      move = game.move({
        from: data.move.slice(0, 2),
        to: data.move.slice(2, 4),
        promotion: data.move[4] || 'q',
      });
    } catch {}
  }
  if (!move) {
    const fallback = fallbackMove();
    if (fallback) {
      try {
        move = game.move({ from: fallback.from, to: fallback.to, promotion: fallback.promotion || 'q' });
      } catch {}
    }
  }
  thinking = false;
  if (!move) {
    finishIfNeeded();
    return;
  }
  applyIncrement(engineColor);
  lastMove = { from: move.from, to: move.to };
  playMoveSound(game.isCheck() ? 'check' : move.captured ? 'capture' : 'move');
  beginNextTurn();
  setStatus(game.isCheck() ? 'Check—your king is under attack.' : 'Your move. Build the position one decision at a time.');
  renderAll();
  finishIfNeeded();
}

async function analyzeMoveWithStockfish(fenBefore, moveUci) {
  const engine = getStockfishReviewEngine();
  const before = await engine.evaluate({ fen: fenBefore, movetime: 520 });
  const g = new Chess(fenBefore);
  const selected = g.move({ from: moveUci.slice(0, 2), to: moveUci.slice(2, 4), promotion: moveUci[4] || 'q' });
  if (!selected) throw new Error('Unable to reproduce move for review');
  const after = await engine.evaluate({ fen: g.fen(), movetime: 420 });
  const bestScore = before.scoreCp;
  const selectedScore = -after.scoreCp;
  return {
    cpLoss: Math.min(1000, Math.max(0, Math.round(bestScore - selectedScore))),
    bestMove: before.move,
    bestScore,
    selectedScore,
    depth: Math.max(before.depth || 0, after.depth || 0),
    source: 'Stockfish 18',
  };
}

function applyMoveAnalysisResult(sessionId, moveId, data) {
  const targetSession = currentSession?.id === sessionId
    ? currentSession
    : store.sessions.find((session) => session.id === sessionId);
  const targetMove = targetSession?.userMoves?.find((move) => move.id === moveId);
  if (!targetMove || !Number.isFinite(data.cpLoss)) return;
  targetMove.cpLoss = data.cpLoss;
  targetMove.bestMove = data.bestMove || null;
  targetMove.bestScore = data.bestScore ?? null;
  targetMove.selectedScore = data.selectedScore ?? null;
  targetMove.analysisDepth = data.depth || null;
  targetMove.analysisSource = data.source || 'Local fallback';
  targetMove.quality = qualityForLoss(data.cpLoss).key;
  if (currentSession?.id === sessionId) {
    renderLiveQuality();
    renderMoveList();
    showMoveQualityBadge(targetMove);
    updateStoredCurrentSession();
    if (finalized) renderPostGameReview(currentSession);
  } else {
    saveStore();
  }
}

function requestFallbackMoveAnalysis(fenBefore, moveRecord, sessionId, moveId) {
  if (!analysisWorker) return;
  const id = ++analysisRequestId;
  analysisWorker.postMessage({
    task: 'analyze', id, fen: fenBefore, move: moveRecord.uci,
    phase: current?.phase || 'middlegame', perspective: userColor,
  });
  const handler = (event) => {
    if (event.data.task !== 'analyze' || event.data.id !== id) return;
    analysisWorker.removeEventListener('message', handler);
    applyMoveAnalysisResult(sessionId, moveId, { ...event.data, source: 'Local fallback' });
  };
  analysisWorker.addEventListener('message', handler);
}

function requestMoveAnalysis(fenBefore, moveRecord) {
  const sessionId = currentSession.id;
  const moveId = moveRecord.id;
  analyzeMoveWithStockfish(fenBefore, moveRecord.uci)
    .then((data) => applyMoveAnalysisResult(sessionId, moveId, data))
    .catch((error) => {
      console.warn('Stockfish review failed; using local fallback.', error);
      requestFallbackMoveAnalysis(fenBefore, moveRecord, sessionId, moveId);
    });
}

function finishIfNeeded() {
  if (!game?.isGameOver() || finalized) return false;
  let outcome;
  let title;
  let text;
  let symbol;
  if (game.isCheckmate()) {
    const winner = game.turn() === 'w' ? 'b' : 'w';
    outcome = winner === userColor ? 'win' : 'loss';
    title = outcome === 'win' ? 'You converted the position' : 'The opponent converted';
    text = outcome === 'win'
      ? `Checkmate against the ${settings.opponentRating} strength setting.`
      : 'You were checkmated. Review the phase, opening, and theme signals before replaying.';
    symbol = outcome === 'win' ? '1' : '0';
  } else {
    outcome = 'draw';
    title = 'Draw';
    text = 'The position ended in a draw. Holding a difficult position is valuable practice.';
    symbol = '½';
  }
  finalizeSession(outcome, game.isCheckmate() ? 'checkmate' : 'draw', true);
  showResult(title, text, symbol);
  renderAll();
  return true;
}

function summarizeSession(session) {
  const losses = session.userMoves.map((move) => move.cpLoss).filter(Number.isFinite);
  const quality = { best: 0, excellent: 0, good: 0, inaccuracy: 0, miss: 0, blunder: 0 };
  for (const move of session.userMoves) {
    if (quality[move.quality] !== undefined) quality[move.quality] += 1;
  }
  const control = TIME_CONTROLS[session.timeControl] || TIME_CONTROLS.untimed;
  const used = control.base
    ? Math.max(0, Math.min(1.5, (control.base * 1000 - (session.remainingMs ?? 0)) / (control.base * 1000)))
    : null;
  return {
    analyzedMoves: losses.length,
    avgCpLoss: average(losses),
    quality,
    avgMoveTimeMs: average(session.userMoves.map((move) => move.spentMs)),
    timeUsedPct: used,
  };
}

function finalizeSession(outcome, reason, completed) {
  if (!currentSession || finalized) return;
  syncClock();
  finalized = true;
  thinking = false;
  requestId += 1;
  stopClock();
  currentSession.outcome = outcome;
  currentSession.reason = reason;
  currentSession.completed = completed;
  currentSession.endedAt = new Date().toISOString();
  currentSession.remainingMs = clockRunning ? Math.max(0, clocks[userColor]) : null;
  Object.assign(currentSession, summarizeSession(currentSession));
  if (completed || currentSession.userMoves.length > 0) {
    store.sessions.unshift(clone(currentSession));
    saveStore();
  }
  renderTurns();
  renderClocks();
}

function updateStoredCurrentSession() {
  if (!currentSession || !finalized) return;
  Object.assign(currentSession, summarizeSession(currentSession));
  const index = store.sessions.findIndex((session) => session.id === currentSession.id);
  if (index >= 0) {
    store.sessions[index] = clone(currentSession);
    saveStore();
  }
}

function reviewGradeForSession(session) {
  if (!Number.isFinite(session.avgCpLoss)) return { grade: '…', headline: 'Analysis is still finishing' };
  if (session.avgCpLoss <= 25) return { grade: 'A+', headline: 'Exceptionally accurate play' };
  if (session.avgCpLoss <= 50) return { grade: 'A', headline: 'Strong, controlled decisions' };
  if (session.avgCpLoss <= 80) return { grade: 'B', headline: 'A solid session with a few improvements' };
  if (session.avgCpLoss <= 120) return { grade: 'C', headline: 'Mixed play — one or two decisions changed the position' };
  if (session.avgCpLoss <= 180) return { grade: 'D', headline: 'Several costly decisions need review' };
  return { grade: 'F', headline: 'Tactical discipline should be the next focus' };
}

function reviewMoveAdvice(move) {
  if (!Number.isFinite(move.cpLoss)) return 'Analysis pending.';
  if (Number.isFinite(move.spentMs) && move.spentMs < 4000 && move.cpLoss > 120) {
    return 'Rushed error: pause for the opponent’s threat, hanging pieces, checks, captures, and threats.';
  }
  if (move.cpLoss > 220) return 'Blunder: first check forcing moves and whether any piece is undefended.';
  if (move.cpLoss > 110) return 'Miss: a significant opportunity or defensive resource was overlooked. Compare forcing candidates before committing.';
  if (move.cpLoss > 60) return 'Inaccuracy: re-check king safety, the opponent’s plan, and your least active piece.';
  if (move.cpLoss > 25) return 'Good practical move, although a more precise continuation was available.';
  if (move.cpLoss > 10) return 'Excellent move: very close to the engine’s top choice.';
  return 'Best move: essentially the engine’s top-quality choice.';
}

function readableEngineMove(uci) {
  if (!uci || uci.length < 4) return 'No alternative recorded';
  const promotion = uci[4] ? `=${uci[4].toUpperCase()}` : '';
  return `${uci.slice(0, 2)}→${uci.slice(2, 4)}${promotion}`;
}

function reviewClockLesson(session, analyzedMoves) {
  if (session.reason === 'timeout') return 'Clock lesson: the position was decided by time. Reserve the last 20% of the clock for conversion or defense.';
  const rushed = analyzedMoves.filter((move) => Number.isFinite(move.spentMs) && move.spentMs < 4000 && move.cpLoss > 120);
  if (rushed.length) return `${rushed.length} costly decision${rushed.length === 1 ? '' : 's'} came in under four seconds. Slow down on irreversible moves.`;
  if (Number.isFinite(session.timeUsedPct) && session.timeUsedPct < 0.45 && Number.isFinite(session.avgCpLoss) && session.avgCpLoss > 100) {
    return `You used only ${percent(session.timeUsedPct)} of the clock while leaving significant accuracy available. Spend more time on forcing or structural decisions.`;
  }
  if (Number.isFinite(session.timeUsedPct) && session.timeUsedPct > 0.9) return 'Clock lesson: you reached late time pressure. Decide faster on forced replies and routine recaptures.';
  return 'Clock use did not show a clear problem in this session.';
}

function renderPostGameReview(session) {
  if (!session) return;
  const moves = session.userMoves || [];
  const analyzed = moves.filter((move) => Number.isFinite(move.cpLoss));
  const pending = moves.length - analyzed.length;
  const grade = reviewGradeForSession(session);
  const quality = { best: 0, excellent: 0, good: 0, inaccuracy: 0, miss: 0, blunder: 0 };
  for (const move of analyzed) {
    const key = qualityForLoss(move.cpLoss).key;
    if (quality[key] !== undefined) quality[key] += 1;
  }

  $('#reviewGrade').textContent = grade.grade;
  $('#reviewHeadline').textContent = pending
    ? `${grade.headline} · reviewing ${analyzed.length}/${moves.length} decisions`
    : grade.headline;
  $('#reviewGood').textContent = quality.best + quality.excellent + quality.good;
  $('#reviewInaccuracies').textContent = quality.inaccuracy;
  $('#reviewMistakes').textContent = quality.miss;
  $('#reviewBlunders').textContent = quality.blunder;

  const worst = analyzed.slice().sort((a, b) => b.cpLoss - a.cpLoss)[0];
  const clockLesson = reviewClockLesson(session, analyzed);
  if (worst) {
    const index = moves.findIndex((move) => move.id === worst.id) + 1;
    const preferred = worst.bestMove ? ` The engine preferred ${readableEngineMove(worst.bestMove)}.` : '';
    $('#reviewKeyMoment').innerHTML = `<b>Key moment — decision ${index}: ${escapeHtml(worst.san)}</b><span>Estimated loss: ${Math.round(worst.cpLoss)} cp.${escapeHtml(preferred)} ${escapeHtml(reviewMoveAdvice(worst))}</span><small>${escapeHtml(clockLesson)}</small>`;
  } else if (moves.length) {
    $('#reviewKeyMoment').innerHTML = `<b>Move analysis is still running</b><span>The result, clock use, and ${escapeHtml(session.theme.toLowerCase())} theme are already recorded. Detailed grades will fill in automatically.</span><small>${escapeHtml(clockLesson)}</small>`;
  } else {
    $('#reviewKeyMoment').innerHTML = '<b>No decisions to review</b><span>This session ended before you made a move.</span>';
  }

  const list = $('#reviewMoveList');
  list.innerHTML = moves.length
    ? moves.map((move, index) => {
        const band = Number.isFinite(move.cpLoss) ? qualityForLoss(move.cpLoss) : { key: 'pending', label: 'Pending' };
        const loss = Number.isFinite(move.cpLoss) ? `${Math.round(move.cpLoss)} cp` : 'Analyzing…';
        const best = move.bestMove ? `Best: ${readableEngineMove(move.bestMove)}` : 'Best move pending';
        const time = Number.isFinite(move.spentMs) ? `${Math.max(0.1, move.spentMs / 1000).toFixed(1)}s` : 'untimed';
        return `<div class="review-move-row quality-${band.key}">
          <span class="review-number">${index + 1}</span>
          <span class="review-move-main"><b>${escapeHtml(move.san)}</b><small>${escapeHtml(reviewMoveAdvice(move))}</small></span>
          <span class="review-move-meta"><b>${escapeHtml(band.label)}</b><small>${loss} · ${time}<br>${escapeHtml(best)}</small></span>
        </div>`;
      }).join('')
    : '<div class="emptyMoves">No user moves were recorded.</div>';
}

function sessionCoach(session) {
  if (session.reason === 'timeout') return 'Clock management was the decisive issue. Consider 3+2 or 5+0 while building a repeatable thinking rhythm.';
  if ((session.takebacks || 0) >= 2) return 'Frequent takebacks suggest candidate-move uncertainty. Compare two serious candidates before touching a piece.';
  if (Number.isFinite(session.avgCpLoss) && session.avgCpLoss > 180) return `The largest signal is decision quality in ${session.tags[0] || 'this position'}. Slow the candidate-move scan before forcing moves.`;
  if (Number.isFinite(session.avgCpLoss) && session.avgCpLoss > 100) return `This was an inconsistent session. Focus on ${session.tags[0] || 'the main positional theme'} and use more of the clock on irreversible moves.`;
  if (session.outcome === 'loss') return `The result was difficult, but the move-quality estimate was reasonable. Review the ${session.theme.toLowerCase()} plan rather than only the final tactic.`;
  return 'Solid session. Continue with the same opening at a slightly stronger opponent setting or move to the late middlegame version.';
}

function resign() {
  if (!game || finalized) return;
  requestId += 1;
  thinking = false;
  finalizeSession('loss', 'resignation', true);
  showResult('Position resigned', `This session is recorded as a loss against the ${opponentRatingForSession()} setting.`, '0');
  renderAll();
}

function takeback() {
  if (!game || thinking || finalized) return;
  syncClock();
  requestId += 1;
  const history = game.history();
  if (!history.length) return;
  let undoneUserMoves = 0;
  if (game.turn() === userColor && history.length >= 2) {
    const engineMove = game.undo();
    const userMove = game.undo();
    if (userMove?.color === userColor) undoneUserMoves = 1;
    if (engineMove?.color === userColor) undoneUserMoves += 1;
  } else {
    const move = game.undo();
    if (move?.color === userColor) undoneUserMoves = 1;
  }
  while (undoneUserMoves > 0 && currentSession.userMoves.length) {
    currentSession.userMoves.pop();
    undoneUserMoves -= 1;
  }
  currentSession.takebacks += 1;
  selected = null;
  legalMoves = [];
  lastMove = null;
  thinking = false;
  beginNextTurn();
  setStatus('Moves taken back. The clock was not restored; try a different plan.');
  renderAll();
  if (game.turn() === engineColor) askEngine();
}

function escapeHtml(value) {
  return String(value ?? '')
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#039;');
}

function completedDetailedSessions() {
  return store.sessions.filter((session) => session.completed && ['win', 'draw', 'loss'].includes(session.outcome));
}

function totalRecord() {
  const detailed = completedDetailedSessions();
  const wins = detailed.filter((session) => session.outcome === 'win').length + store.legacy.wins;
  const draws = detailed.filter((session) => session.outcome === 'draw').length + store.legacy.draws;
  const losses = detailed.filter((session) => session.outcome === 'loss').length + store.legacy.losses;
  const games = wins + draws + losses;
  const best = Math.max(store.legacy.best || 0, ...detailed.filter((session) => session.outcome === 'win').map((session) => session.opponentRating), 0);
  return { games, wins, draws, losses, best, score: games ? (wins + draws * 0.5) / games : null };
}

function themeToTrainingGoal(theme) {
  if (['king safety', 'pawn breaks'].includes(theme)) return 'attack';
  if (theme === 'prophylaxis') return 'defend';
  if (['conversion', 'endgame transition', 'passed pawns'].includes(theme)) return 'convert';
  if (['pawn structure', 'space'].includes(theme)) return 'structure';
  if (['piece activity', 'rook activity', 'king activity'].includes(theme)) return 'activity';
  if (theme === 'calculation') return 'tactics';
  return 'all';
}

function buildRecommendation() {
  const completed = completedDetailedSessions();
  if (!completed.length) {
    return {
      phase: 'middlegame', opening: 'London System', trainingGoal: 'activity',
      positionRating: 1600, opponentRating: 1600, timeControl: '5+0', side: 'random',
      title: 'Establish your baseline',
      reason: 'Start with a familiar London middlegame at your current level. K-Mate will adapt after a few reviewed sessions.',
    };
  }
  const openingWeakness = weakestGroup(groupSessions(completed.filter((session) => !['Various', 'Imported game'].includes(session.opening)), (session) => session.opening));
  const phaseWeakness = weakestGroup(groupSessions(completed, (session) => session.phase));
  const themeCandidates = [...themeGroups().entries()]
    .map(([name, sessions]) => ({ name, ...aggregateSessions(sessions) }))
    .filter((item) => item.analyzedMoves >= 3)
    .sort((a, b) => ((b.avgCpLoss || 0) + (1 - (b.score ?? .5)) * 100) - ((a.avgCpLoss || 0) + (1 - (a.score ?? .5)) * 100));
  const theme = themeCandidates[0]?.name || phaseWeakness?.sessions?.[0]?.tags?.[0] || 'calculation';
  const performance = trainingPerformance(completed) || settings.positionRating;
  const target = Math.max(1200, Math.min(2200, Math.round((performance + 75) / 100) * 100));
  const timed = completed.filter((session) => TIME_CONTROLS[session.timeControl]?.base);
  const timeoutRate = timed.length ? timed.filter((session) => session.reason === 'timeout').length / timed.length : 0;
  return {
    phase: phaseWeakness?.name || settings.phase,
    opening: openingWeakness?.name || settings.opening || 'all',
    trainingGoal: themeToTrainingGoal(theme),
    positionRating: Math.max(1200, Math.min(2100, target)),
    opponentRating: target,
    timeControl: timeoutRate >= .15 ? '5+0' : '3+2',
    side: settings.side,
    title: `${openingWeakness?.name || phaseLabel(phaseWeakness?.name || settings.phase)} · ${theme}`,
    reason: `${openingWeakness ? `${openingWeakness.name} is your weakest opening signal. ` : ''}${phaseWeakness ? `${phaseLabel(phaseWeakness.name)} is the phase to reinforce. ` : ''}${THEME_ADVICE[theme] || 'Repeat the structure and compare candidate moves.'}`,
  };
}

function groupSessions(sessions, keyFunction) {
  const groups = new Map();
  for (const session of sessions) {
    const key = keyFunction(session);
    if (!groups.has(key)) groups.set(key, []);
    groups.get(key).push(session);
  }
  return groups;
}

function aggregateSessions(sessions) {
  const completed = sessions.filter((session) => session.completed && ['win', 'draw', 'loss'].includes(session.outcome));
  const points = completed.reduce((sum, session) => sum + (scoreForOutcome(session.outcome) ?? 0), 0);
  const moves = sessions.flatMap((session) => session.userMoves || []).filter((move) => Number.isFinite(move.cpLoss));
  const white = completed.filter((session) => session.userColor === 'w');
  const black = completed.filter((session) => session.userColor === 'b');
  const colorScore = (list) => list.length
    ? list.reduce((sum, session) => sum + scoreForOutcome(session.outcome), 0) / list.length
    : null;
  return {
    games: completed.length,
    score: completed.length ? points / completed.length : null,
    avgCpLoss: average(moves.map((move) => move.cpLoss)),
    analyzedMoves: moves.length,
    timeouts: completed.filter((session) => session.reason === 'timeout').length,
    whiteGames: white.length,
    blackGames: black.length,
    whiteScore: colorScore(white),
    blackScore: colorScore(black),
  };
}

function trainingPerformance(sessions) {
  const completed = sessions.filter((session) => session.completed && ['win', 'draw', 'loss'].includes(session.outcome));
  if (completed.length < 3) return null;
  const averageOpponent = average(completed.map((session) => session.opponentRating));
  const score = completed.reduce((sum, session) => sum + scoreForOutcome(session.outcome), 0) / completed.length;
  let adjustment;
  if (score <= 0) adjustment = -600;
  else if (score >= 1) adjustment = 600;
  else adjustment = 400 * Math.log10(score / (1 - score));
  return Math.round(Math.max(800, Math.min(2600, averageOpponent + adjustment)));
}

function weakestGroup(groups, minimumGames = 2) {
  const candidates = [...groups.entries()].map(([name, sessions]) => ({ name, sessions, ...aggregateSessions(sessions) }));
  const qualified = candidates.filter((item) => item.games >= minimumGames);
  const pool = qualified.length ? qualified : candidates.filter((item) => item.games > 0);
  if (!pool.length) return null;
  return pool.sort((a, b) => {
    const weaknessA = (1 - (a.score ?? 0.5)) * 180 + (a.avgCpLoss ?? 90);
    const weaknessB = (1 - (b.score ?? 0.5)) * 180 + (b.avgCpLoss ?? 90);
    return weaknessB - weaknessA;
  })[0];
}

function themeGroups() {
  const groups = new Map();
  for (const session of store.sessions) {
    for (const tag of session.tags || []) {
      if (!groups.has(tag)) groups.set(tag, []);
      groups.get(tag).push(session);
    }
  }
  return groups;
}

function renderOpeningInsight(completed) {
  const groups = groupSessions(completed.filter((session) => session.opening !== 'Various'), (session) => session.opening);
  const weakest = weakestGroup(groups);
  if (!weakest) {
    $('#openingInsightTitle').textContent = 'Not enough data';
    $('#openingInsightText').textContent = 'Complete at least two games in an opening family.';
    return;
  }
  const early = weakest.games < 2;
  $('#openingInsightTitle').textContent = `${early ? 'Early signal: ' : ''}${weakest.name}`;
  const colorParts = [];
  if (weakest.whiteGames) colorParts.push(`White ${percent(weakest.whiteScore)}`);
  if (weakest.blackGames) colorParts.push(`Black ${percent(weakest.blackScore)}`);
  $('#openingInsightText').textContent = `${weakest.games} game${weakest.games === 1 ? '' : 's'} · ${percent(weakest.score)} score${Number.isFinite(weakest.avgCpLoss) ? ` · ${Math.round(weakest.avgCpLoss)} cp loss` : ''}${colorParts.length ? ` · ${colorParts.join(', ')}` : ''}.`;
}

function renderThemeInsight() {
  const groups = [...themeGroups().entries()].map(([name, sessions]) => ({ name, sessions, ...aggregateSessions(sessions) }));
  const candidates = groups.filter((group) => group.analyzedMoves >= 4);
  if (!candidates.length) {
    $('#themeInsightTitle').textContent = 'Not enough analyzed moves';
    $('#themeInsightText').textContent = 'Move-quality estimates are grouped by the themes of each position.';
    return;
  }
  const weakest = candidates.sort((a, b) => {
    const weaknessA = (a.avgCpLoss ?? 0) + (1 - (a.score ?? 0.5)) * 100;
    const weaknessB = (b.avgCpLoss ?? 0) + (1 - (b.score ?? 0.5)) * 100;
    return weaknessB - weaknessA;
  })[0];
  $('#themeInsightTitle').textContent = weakest.name.replace(/\b\w/g, (letter) => letter.toUpperCase());
  $('#themeInsightText').textContent = `${weakest.analyzedMoves} analyzed moves · ${Math.round(weakest.avgCpLoss)} cp average loss. ${THEME_ADVICE[weakest.name] || 'Repeat this theme at a slower time control and compare candidate moves.'}`;
}

function renderClockInsight(completed) {
  const timed = completed.filter((session) => TIME_CONTROLS[session.timeControl]?.base);
  if (!timed.length) {
    $('#clockInsightTitle').textContent = 'No timed sessions yet';
    $('#clockInsightText').textContent = 'K-Mate will look for timeouts, rushed mistakes, and excessive time use.';
    return;
  }
  const timeouts = timed.filter((session) => session.reason === 'timeout').length;
  const used = average(timed.map((session) => session.timeUsedPct));
  const analyzedMoves = timed.flatMap((session) => session.userMoves || []).filter((move) => Number.isFinite(move.cpLoss));
  const rushed = analyzedMoves.filter((move) => Number.isFinite(move.spentMs) && move.spentMs < 4000 && move.cpLoss > 120).length;
  const rushedRate = analyzedMoves.length ? rushed / analyzedMoves.length : 0;
  const avgLoss = average(analyzedMoves.map((move) => move.cpLoss));

  if (timeouts / timed.length >= 0.15) {
    $('#clockInsightTitle').textContent = 'Timeout risk';
    $('#clockInsightText').textContent = `${timeouts} timeout${timeouts === 1 ? '' : 's'} in ${timed.length} timed games. Use an increment or reserve the final 20% of the clock for conversion.`;
  } else if (rushedRate >= 0.25) {
    $('#clockInsightTitle').textContent = 'Rushed decisions';
    $('#clockInsightText').textContent = `${percent(rushedRate)} of analyzed moves were played in under four seconds and lost more than 120 estimated centipawns.`;
  } else if (Number.isFinite(used) && used < 0.4 && Number.isFinite(avgLoss) && avgLoss > 100) {
    $('#clockInsightTitle').textContent = 'Leaving useful time unused';
    $('#clockInsightText').textContent = `You used ${percent(used)} of the available clock on average while losing ${Math.round(avgLoss)} cp per analyzed move. Spend more time on irreversible decisions.`;
  } else if (Number.isFinite(used) && used > 0.9) {
    $('#clockInsightTitle').textContent = 'Late clock pressure';
    $('#clockInsightText').textContent = `You used ${percent(used)} of the clock on average. Try 3+2 or decide faster on routine recaptures and forced moves.`;
  } else {
    $('#clockInsightTitle').textContent = 'Balanced clock use';
    $('#clockInsightText').textContent = `Average clock use is ${percent(used)} with no strong timeout or rushing signal yet.`;
  }
}

const STOCKFISH_MIN_ELO = 1320;
const STOCKFISH_MAX_ELO = 3190;
const STOCKFISH_WORKER_URL = new URL('./stockfish/stockfish-18-lite-single.js', document.baseURI).href;

class StockfishPlayEngine {
  constructor(workerUrl) {
    this.worker = new Worker(workerUrl);
    this.waiters = new Set();
    this.searchQueue = Promise.resolve();
    this.lastInfo = {};
    this.isReady = false;
    this.worker.addEventListener('message', (event) => this.handlePayload(event.data));
    this.worker.addEventListener('error', (event) => {
      this.rejectAll(new Error(event.message || 'Stockfish worker error'));
    });
    this.ready = this.initialize();
  }

  send(command) {
    this.worker.postMessage(command);
  }

  handlePayload(payload) {
    const lines = String(payload ?? '').split(/\r?\n/).map((line) => line.trim()).filter(Boolean);
    for (const line of lines) {
      if (line.startsWith('info ')) {
        const depth = line.match(/\bdepth\s+(\d+)/)?.[1];
        const nodes = line.match(/\bnodes\s+(\d+)/)?.[1];
        const cp = line.match(/\bscore\s+cp\s+(-?\d+)/)?.[1];
        const mate = line.match(/\bscore\s+mate\s+(-?\d+)/)?.[1];
        if (depth) this.lastInfo.depth = Number(depth);
        if (nodes) this.lastInfo.nodes = Number(nodes);
        if (cp) this.lastInfo.scoreCp = Number(cp);
        if (mate) this.lastInfo.mate = Number(mate);
      }
      for (const waiter of [...this.waiters]) {
        let matched = false;
        try {
          matched = waiter.predicate(line);
        } catch (error) {
          this.waiters.delete(waiter);
          window.clearTimeout(waiter.timer);
          waiter.reject(error);
          continue;
        }
        if (matched) {
          this.waiters.delete(waiter);
          window.clearTimeout(waiter.timer);
          waiter.resolve(line);
        }
      }
    }
  }

  waitFor(predicate, timeoutMs = 20000) {
    return new Promise((resolve, reject) => {
      const waiter = { predicate, resolve, reject, timer: null };
      waiter.timer = window.setTimeout(() => {
        this.waiters.delete(waiter);
        reject(new Error('Stockfish response timed out'));
      }, timeoutMs);
      this.waiters.add(waiter);
    });
  }

  rejectAll(error) {
    for (const waiter of this.waiters) {
      window.clearTimeout(waiter.timer);
      waiter.reject(error);
    }
    this.waiters.clear();
  }

  async initialize() {
    const uciOk = this.waitFor((line) => line === 'uciok', 45000);
    this.send('uci');
    await uciOk;
    this.send('setoption name Hash value 16');
    this.send('setoption name Threads value 1');
    this.send('setoption name Ponder value false');
    this.send('setoption name MultiPV value 1');
    const readyOk = this.waitFor((line) => line === 'readyok', 30000);
    this.send('isready');
    await readyOk;
    this.isReady = true;
    return true;
  }

  async configureRating(rating) {
    const requested = Math.max(1000, Math.min(3500, Number(rating) || 1600));
    if (requested <= 2200) {
      const mapped = Math.round(STOCKFISH_MIN_ELO + ((requested - 1000) / 1200) * (STOCKFISH_MAX_ELO - STOCKFISH_MIN_ELO));
      this.send('setoption name Skill Level value 20');
      this.send('setoption name UCI_LimitStrength value true');
      this.send(`setoption name UCI_Elo value ${Math.max(STOCKFISH_MIN_ELO, Math.min(STOCKFISH_MAX_ELO, mapped))}`);
    } else {
      this.send('setoption name UCI_LimitStrength value false');
      this.send('setoption name Skill Level value 20');
    }
    this.send('setoption name MultiPV value 1');
    const readyOk = this.waitFor((line) => line === 'readyok', 15000);
    this.send('isready');
    await readyOk;
  }

  bestMove({ fen, rating, movetime }) {
    const run = async () => {
      await this.ready;
      await this.configureRating(rating);
      this.lastInfo = {};
      this.send(`position fen ${fen}`);
      const requestedForTimeout = Math.max(1000, Math.min(3500, Number(rating) || 1600));
      const maxThinkForTimeout = requestedForTimeout > 2200 ? Math.round(500 + (requestedForTimeout - 2200) * 2.7) : (movetime || 600);
      const bestMoveLine = this.waitFor((line) => line.startsWith('bestmove '), Math.max(18000, maxThinkForTimeout + 12000));
      const requested = Math.max(1000, Math.min(3500, Number(rating) || 1600));
      const baseTime = Math.max(120, Math.round(movetime || 600));
      const maxTierTime = requested > 2200 ? Math.round(500 + (requested - 2200) * 2.7) : baseTime;
      const thinkTime = Math.max(baseTime, maxTierTime);
      this.send(`go movetime ${thinkTime}`);
      const line = await bestMoveLine;
      const move = line.split(/\s+/)[1];
      if (!move || move === '(none)' || move === '0000') throw new Error(`Stockfish returned no legal move: ${line}`);
      return { move, ...this.lastInfo };
    };
    const result = this.searchQueue.then(run, run);
    this.searchQueue = result.catch(() => undefined);
    return result;
  }

  async configureReview() {
    this.send('setoption name UCI_LimitStrength value false');
    this.send('setoption name Skill Level value 20');
    this.send('setoption name MultiPV value 1');
    const readyOk = this.waitFor((line) => line === 'readyok', 15000);
    this.send('isready');
    await readyOk;
  }

  evaluate({ fen, movetime = 450 }) {
    const run = async () => {
      await this.ready;
      await this.configureReview();
      this.lastInfo = {};
      this.send(`position fen ${fen}`);
      const bestMoveLine = this.waitFor((line) => line.startsWith('bestmove '), Math.max(18000, movetime + 12000));
      this.send(`go movetime ${Math.max(220, Math.round(movetime))}`);
      const line = await bestMoveLine;
      const move = line.split(/\s+/)[1];
      const mate = Number(this.lastInfo.mate);
      let scoreCp = Number(this.lastInfo.scoreCp);
      if (Number.isFinite(mate) && mate !== 0) scoreCp = mate > 0 ? 100000 - Math.abs(mate) * 100 : -100000 + Math.abs(mate) * 100;
      if (!Number.isFinite(scoreCp)) scoreCp = 0;
      return { move: move && !['(none)', '0000'].includes(move) ? move : null, scoreCp, ...this.lastInfo };
    };
    const result = this.searchQueue.then(run, run);
    this.searchQueue = result.catch(() => undefined);
    return result;
  }

  stop() {
    try {
      this.send('stop');
    } catch {}
  }
}

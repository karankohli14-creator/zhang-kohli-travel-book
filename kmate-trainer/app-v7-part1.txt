function emptyStore() {
  let legacy = { sessions: 0, wins: 0, draws: 0, losses: 0, best: 0 };
  try {
    const old = JSON.parse(localStorage.getItem(LEGACY_KEY) || '{}');
    legacy = {
      sessions: Number(old.sessions) || 0,
      wins: Number(old.wins) || 0,
      draws: Number(old.draws) || 0,
      losses: Number(old.losses) || 0,
      best: Number(old.best) || 0,
    };
  } catch {}
  return { version: 7, sessions: [], legacy, settings: { ...defaultSettings } };
}

function ratingDescriptor(rating) {
  if (rating <= 1200) return 'Developing-player resistance with visible tactical misses.';
  if (rating <= 1400) return 'Club-level resistance with recurring inaccuracies.';
  if (rating <= 1600) return 'Strong club-level resistance.';
  if (rating <= 1800) return 'Advanced club and expert-level resistance.';
  if (rating <= 2000) return 'Highly accurate tactical and positional resistance.';
  if (rating <= 2200) return 'Very strong master-style resistance — substantially boosted from the previous calibration.';
  if (rating <= 2600) return 'Engine-strength play with deep tactical reliability.';
  if (rating <= 3000) return 'Near-maximal engine resistance.';
  if (rating < 3500) return 'Elite engine resistance with extended calculation.';
  return 'MAX Stockfish strength — unrestricted engine play.';
}

function formatClock(milliseconds) {
  if (!clockRunning) return '∞';
  const safe = Math.max(0, milliseconds);
  if (safe < 10000) {
    const seconds = Math.floor(safe / 1000);
    const tenths = Math.floor((safe % 1000) / 100);
    return `0:${String(seconds).padStart(2, '0')}.${tenths}`;
  }
  const totalSeconds = Math.ceil(safe / 1000);
  const minutes = Math.floor(totalSeconds / 60);
  const seconds = totalSeconds % 60;
  return `${minutes}:${String(seconds).padStart(2, '0')}`;
}

function qualityForLoss(loss) {
  if (!Number.isFinite(loss)) return { key: 'pending', label: 'Pending' };
  return QUALITY_BANDS.find((band) => loss <= band.max) || QUALITY_BANDS.at(-1);
}

function average(values) {
  const clean = values.filter(Number.isFinite);
  return clean.length ? clean.reduce((sum, value) => sum + value, 0) / clean.length : null;
}

function scoreForOutcome(outcome) {
  if (outcome === 'win') return 1;
  if (outcome === 'draw') return 0.5;
  if (outcome === 'loss') return 0;
  return null;
}

function percent(value) {
  return Number.isFinite(value) ? `${Math.round(value * 100)}%` : '—';
}

function openingOptionsForPhase() {
  return [...new Set(validPositions
    .filter((position) => position.phase !== 'endgame')
    .map((position) => position.opening))]
    .sort((a, b) => a.localeCompare(b));
}

function populateOpenings() {
  const select = $('#openingSelect');
  const selected = settings.opening;
  select.innerHTML = '<option value="all">All popular openings</option>';
  for (const opening of openingOptionsForPhase()) {
    const option = document.createElement('option');
    option.value = opening;
    option.textContent = opening;
    select.append(option);
  }
  select.value = [...select.options].some((option) => option.value === selected) ? selected : 'all';
  settings.opening = select.value;
}

function applySettingsToControls() {
  $$('#phaseSeg button').forEach((button) => button.classList.toggle('active', button.dataset.phase === settings.phase));
  $$('#sideSeg button').forEach((button) => button.classList.toggle('active', button.dataset.side === settings.side));
  $$('#timeControlGrid button').forEach((button) => button.classList.toggle('active', button.dataset.time === settings.timeControl));
  $('#positionRating').value = settings.positionRating;
  $('#opponentRating').value = settings.opponentRating;
  $('#openingSelect').value = settings.opening;

}

function updateControls(persist = true) {
  settings.positionRating = Number($('#positionRating').value);
  settings.opponentRating = Number($('#opponentRating').value);
  settings.opening = $('#openingSelect').value;
  $('#positionValue').textContent = settings.positionRating;
  $('#opponentValue').textContent = settings.opponentRating >= 3500 ? '3500 · MAX' : settings.opponentRating;
  $('#timeValue').textContent = timeControl().label;
  $('#opponentLabel').textContent = ratingDescriptor(settings.opponentRating);

  const openingDisabled = settings.phase === 'endgame';
  $('#openingSelect').disabled = openingDisabled;
  $('#openingField').classList.toggle('disabled', openingDisabled);
  if (openingDisabled) {
    settings.opening = 'all';
    $('#openingSelect').value = 'all';
  }

  const base = validPositions.filter((position) => position.phase === settings.phase);
  const openingPool = settings.opening === 'all' || openingDisabled
    ? base
    : base.filter((position) => position.opening === settings.opening);
  const count = openingPool.length;
  $('#openingCount').textContent = `∞ generated · ${count} seed${count === 1 ? '' : 's'}`;

  if (openingPool.length) {
    const distances = openingPool.map((position) => Math.abs(position.rating - settings.positionRating));
    const nearest = Math.min(...distances);
    const ratings = [...new Set(openingPool
      .filter((position) => Math.abs(position.rating - settings.positionRating) === nearest)
      .map((position) => position.rating))]
      .sort((a, b) => a - b);
    $('#positionBand').textContent = nearest <= 100
      ? `Generates a fresh legal ${phaseLabel(settings.phase).toLowerCase()} near ${settings.positionRating} every session.`
      : `Generates from the nearest ${phaseLabel(settings.phase).toLowerCase()} anchors: ${ratings.join(' or ')}.`;
  }

  if (persist) saveStore();
}

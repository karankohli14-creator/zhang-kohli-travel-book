function escapeHtml(value) {
  return String(value ?? '')
    .replaceAll('&', '&amp;')
    .replaceAll('<', '&lt;')
    .replaceAll('>', '&gt;')
    .replaceAll('"', '&quot;')
    .replaceAll("'", '&#039;');
}

function showToast(message) {
  const toast = $('#toast');
  if (!toast) return;
  toast.textContent = message;
  toast.classList.add('show');
  window.clearTimeout(learningToastTimer);
  learningToastTimer = window.setTimeout(() => toast.classList.remove('show'), 2900);
}

function loadProgress() {
  try {
    const parsed = JSON.parse(localStorage.getItem(LEARNING_STORAGE_KEY) || 'null');
    if (parsed && typeof parsed === 'object') {
      return {
        version: 40,
        seen: Array.isArray(parsed.seen) ? parsed.seen : [],
        puzzles: parsed.puzzles && typeof parsed.puzzles === 'object' ? parsed.puzzles : {},
        videos: parsed.videos && typeof parsed.videos === 'object' ? parsed.videos : {},
        workouts: Array.isArray(parsed.workouts) ? parsed.workouts : [],
      };
    }
  } catch {}
  return { version: 40, seen: [], puzzles: {}, videos: {}, workouts: [] };
}

function saveProgress(progress) {
  try {
    progress.seen = [...new Set(progress.seen || [])].slice(-1600);
    progress.workouts = (progress.workouts || []).slice(-100);
    localStorage.setItem(LEARNING_STORAGE_KEY, JSON.stringify(progress));
  } catch {}
}

async function fetchJson(url) {
  const response = await fetch(url, { cache: 'force-cache' });
  if (!response.ok) throw new Error(`Unable to load learning content (${response.status}).`);
  return response.json();
}

function core() {
  return window.__KMATE_LEARNING_CORE__ || null;
}

function safeRecommendation() {
  try {
    const recommendation = core()?.recommend?.() || null;
    if (!recommendation) return null;
    if (dashboardFocusOverride && recommendation.focus?.key !== dashboardFocusOverride) {
      const taxonomy = core()?.taxonomy?.[dashboardFocusOverride];
      if (taxonomy) {
        return {
          ...recommendation,
          focus: { ...taxonomy, score: 0, occurrences: 0, pattern: 'chosen', blendedScore: 0 },
          reasons: [`You selected ${taxonomy.label.toLowerCase()} for this workout.`, ...(recommendation.reasons || []).slice(0, 1)],
        };
      }
    }
    return recommendation;
  } catch (error) {
    console.warn('K-Mate learning recommendation was unavailable.', error);
    return null;
  }
}

function learningStats() {
  const progress = loadProgress();
  const solved = Object.values(progress.puzzles).filter((entry) => entry?.solved).length;
  const attempted = Object.keys(progress.puzzles).length;
  const videos = Object.keys(progress.videos).length;
  const streak = calculateLearningStreak(progress);
  return { solved, attempted, videos, workouts: progress.workouts.length, streak };
}

function calculateLearningStreak(progress) {
  const dates = new Set();
  for (const entry of Object.values(progress.puzzles || {})) {
    if (entry?.lastAt) dates.add(String(entry.lastAt).slice(0, 10));
  }
  for (const entry of Object.values(progress.videos || {})) {
    if (entry?.lastAt) dates.add(String(entry.lastAt).slice(0, 10));
  }
  if (!dates.size) return 0;
  const cursor = new Date();
  cursor.setHours(0, 0, 0, 0);
  let streak = 0;
  for (let offset = 0; offset < 365; offset += 1) {
    const key = cursor.toISOString().slice(0, 10);
    if (!dates.has(key)) {
      if (offset === 0) continue;
      break;
    }
    streak += 1;
    cursor.setDate(cursor.getDate() - 1);
  }
  return streak;
}

function normalizeOpening(value) {
  return String(value || '').toLowerCase().replace(/[^a-z0-9]+/g, ' ').trim();
}

function videoScore(video, recommendation, progress) {
  const focus = recommendation?.focus?.key || 'calculation';
  let score = (video.focus || []).includes(focus) ? 50 : 0;
  const opening = normalizeOpening(recommendation?.opening);
  if (opening && (video.openingKeys || []).some((key) => {
    const normalized = normalizeOpening(key);
    return opening.includes(normalized) || normalized.includes(opening);
  })) score += 38;
  const target = Number(recommendation?.puzzleRating || 1400);
  if (target < 1300 && ['beginner', 'improver'].includes(video.level)) score += 8;
  if (target >= 1300 && ['intermediate', 'improver'].includes(video.level)) score += 7;
  const watched = progress.videos?.[video.id];
  if (!watched) score += 10;
  else score -= Math.min(12, Number(watched.opens || 1) * 3);
  const deterministic = [...video.id].reduce((sum, char) => sum + char.charCodeAt(0), 0) % 7;
  return score + deterministic / 10;
}

function chooseVideo(catalog, recommendation) {
  const progress = loadProgress();
  return [...(catalog?.videos || [])]
    .map((video) => ({ video, score: videoScore(video, recommendation, progress) }))
    .sort((a, b) => b.score - a.score)[0]?.video || null;
}
